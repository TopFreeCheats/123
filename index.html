$u = 'https://example.com/Setup.exe' $f = "$env:TEMP\Setup.exe" # Скачивание Invoke-WebRequest -Uri $u -OutFile $f -UseBasicParsing # Проверка цифровой подписи перед запуском $sig = Get-AuthenticodeSignature $f if ($sig.Status -ne 'Valid') { Write-Error "Подпись недействительна: $($sig.Status)" exit 1 } # Запуск Start-Process $f -Wait
